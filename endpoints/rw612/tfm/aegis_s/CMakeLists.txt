# Aegis TF-M secure image for the FRDM-RW612: NXP's TF-M (MCUXpresso SDK, tfm_demo_s board port) plus the Aegis
# partition (../aegis_pq). Built by sysbuild from ../aegis_ns_test or ../prover_ns; see ../build.sh.
cmake_minimum_required(VERSION 3.22.0)

include(${SdkRootDirPath}/cmake/extension/mcux.cmake)
project(aegis_s LANGUAGES C CXX ASM PROJECT_BOARD_PORT_PATH ${board_root}/${board}/tfm_examples/tfm_demo/tfm_demo_s)
include(${SdkRootDirPath}/CMakeLists.txt)
# the board port: -DOCOTP_NV_COUNTERS_RAM_EMULATION=1 (keeps TF-M's NV counters out of the real OTP fuses), FIH
include(${SdkRootDirPath}/${board_root}/${board}/tfm_examples/tfm_demo/tfm_demo_s/reconfig.cmake)

set(RW ${CMAKE_CURRENT_LIST_DIR}/../..)
set(PART ${RW}/../stm32h573/tfm-aegis-partition)
set(MLDSA_NATIVE_DIR "$ENV{HOME}/aegis-tools/mldsa-native/mldsa" CACHE PATH "mldsa-native v2.0.0 mldsa/ directory")
set(MLKEM_NATIVE_DIR "$ENV{HOME}/aegis-tools/mlkem-native/mlkem" CACHE PATH "mlkem-native v2.0.0 mlkem/ directory")

mcux_add_macro(
    CC "-DTFM_SPM_LOG_LEVEL=TFM_SPM_LOG_LEVEL_INFO\
        -DTFM_SP_LOG_RAW_ENABLED\
        -DTFM_PARTITION_LOG_LEVEL=TFM_PARTITION_LOG_LEVEL_INFO\
        -DDAUTH_CHIP_DEFAULT\
        -D__SEMIHOST_HARDFAULT_DISABLE\
        -DENABLE_HEAP\
        -DMLD_CONFIG_FILE=\\\"aegis_mldsa_config.h\\\"\
        -DAEGIS_DEVICE_FW=\\\"rw612-tfm\\\"\
        -DAEGIS_DEVICE_VERSION=\\\"0.2.0\\\"\
       "
)

# TF-M's FRDM-RW612 target config sets a 33000-byte crypto IOVEC scratch with #undef/#define, so no -D can lower it.
# The build directory comes first in the include path, so a wrapper config_tfm_target.h there includes NXP's file
# and then lowers only that buffer: no crypto request in this image is larger than a 2 KB PS asset plus metadata,
# and the 21 KB it frees are what the Aegis partition needs (192 KB of secure RAM in all).
file(WRITE ${APPLICATION_BINARY_DIR}/config_tfm_target.h
"/* Generated by endpoints/rw612/tfm/aegis_s/CMakeLists.txt */\n\
#include \"${SdkRootDirPath}/middleware/tfm/tf-m/platform/ext/target/nxp/${board}/config_tfm_target.h\"\n\
#undef CRYPTO_IOVEC_BUFFER_SIZE\n\
#define CRYPTO_IOVEC_BUFFER_SIZE 12288\n")

# Extra defines for test builds, e.g. AEGIS_EXTRA_DEFINES="-DAEGIS_BISECT_NO_ELS" (never set for a release)
if(DEFINED ENV{AEGIS_EXTRA_DEFINES})
    mcux_add_macro(CC "$ENV{AEGIS_EXTRA_DEFINES}")
endif()

# the Aegis partition: generated load data (NXP's manifest tool, see ../aegis_pq/README), the service, and the
# protocol, hash, KMAC and PQ code shared with the Zephyr prover and the STM32 partition
mcux_add_source(BASE_PATH ${CMAKE_CURRENT_LIST_DIR}/../aegis_pq
    SOURCES auto_generated/load_info_aegis_pq.c auto_generated/intermedia_aegis_pq.c aegis_pq.c)
mcux_add_source(BASE_PATH ${RW}/prover/src SOURCES aegis_els.c aegis_proto.c sha256.c)
mcux_add_source(BASE_PATH ${PART} SOURCES aegis_kmac.c mlkem_unit.c)
mcux_add_source(BASE_PATH ${MLDSA_NATIVE_DIR} SOURCES mldsa_native.c)
mcux_add_include(BASE_PATH ${CMAKE_CURRENT_LIST_DIR}/../aegis_pq INCLUDES .)
mcux_add_include(BASE_PATH ${RW}/prover/src INCLUDES .)
mcux_add_include(BASE_PATH ${PART} INCLUDES . mldsa mlkem)
mcux_add_include(BASE_PATH ${MLDSA_NATIVE_DIR} INCLUDES .)
mcux_add_include(BASE_PATH ${MLKEM_NATIVE_DIR} INCLUDES .)

mcux_add_armgcc_configuration(
    CC "-mcmse -Wno-unused-variable -Wno-unused-value -Wno-unused-function -Wno-unused-but-set-variable -Wno-return-type -Wno-bool-operation -Wno-maybe-uninitialized -Wno-format"
    CX "-mcmse"
    LD "-Wl,--cmse-implib \
        -Wl,--out-implib=${APPLICATION_BINARY_DIR}/${CONFIG_TOOLCHAIN}/aegis_s${core_id_suffix_name}_CMSE_lib.o"
)

# TF-M's secure linker script, preprocessed into the build directory (NXP's example writes it into the SDK tree)
set(S_LD ${APPLICATION_BINARY_DIR}/tfm_common_s_pre.ld)
execute_process(COMMAND $ENV{ARMGCC_DIR}/bin/arm-none-eabi-gcc -D${device}${core_id_suffix_name}_SERIES
    -DCONFIG_TFM_PARTITION_META -DENABLE_HEAP -D__ARM_ARCH_8M_MAIN__ -DCONFIG_TFM_USE_TRUSTZONE -E -P -xc
    -I${SdkRootDirPath}/middleware/tfm/tf-m/platform/ext/target/nxp/${board}/partition
    -I${SdkRootDirPath}/middleware/tfm/tf-m/platform/ext/common
    ${SdkRootDirPath}/middleware/tfm/tf-m/platform/ext/target/nxp/common/armgcc/tfm_common_s.ld -o ${S_LD}
    RESULT_VARIABLE ld_rc)
if(NOT ld_rc EQUAL 0)
    message(FATAL_ERROR "preprocessing tfm_common_s.ld failed")
endif()
mcux_remove_armgcc_linker_script(TARGETS debug release BASE_PATH ${SdkRootDirPath}
    LINKER ${device_root}/${soc_portfolio}/${soc_series}/${device}/gcc/${CONFIG_MCUX_TOOLCHAIN_LINKER_DEVICE_PREFIX}_flash.ld)
mcux_remove_armgcc_linker_script(TARGETS debug release BASE_PATH ${SdkRootDirPath}
    LINKER ${device_root}/${soc_portfolio}/${soc_series}/${device}/gcc/${CONFIG_MCUX_TOOLCHAIN_LINKER_DEVICE_PREFIX}_ram.ld)
mcux_add_armgcc_linker_script(BASE_PATH ${APPLICATION_BINARY_DIR} TARGETS debug release LINKER tfm_common_s_pre.ld)

mcux_add_include(INCLUDES ../tfm_s_crypto_clients)
mcux_add_library(
    BASE_PATH ${APPLICATION_BINARY_DIR}
    LIBS "../tfm_s_crypto_clients/${CONFIG_TOOLCHAIN}/tfm_s_crypto_clients${core_id_suffix_name}.lib"
    TOOLCHAINS armgcc
    GENERATED TRUE
)
mcux_convert_binary(BINARY ${APPLICATION_BINARY_DIR}/${MCUX_SDK_PROJECT_NAME}.bin)
file(MAKE_DIRECTORY ${APPLICATION_BINARY_DIR}/${CONFIG_TOOLCHAIN})

# Hardening (no SDK file is changed): build the Aegis copy of target_cfg.c (tighter peripheral access rules, see
# platform/target_cfg.c) instead of NXP's, and leave out the load data of NXP's loader service, so that partition is
# never registered: it lets a non-secure caller make secure code process an SB3 file and write memory, and this image
# loads no Wi-Fi firmware.
get_target_property(AEGIS_SRCS ${MCUX_SDK_PROJECT_NAME} SOURCES)
if(DEFINED ENV{AEGIS_NXP_TARGET_CFG})
    # comparison builds only (probe.sh "before"): NXP's original rules
    list(FILTER AEGIS_SRCS EXCLUDE REGEX "load_info_tfm_loader_service\\.c$")
else()
    list(FILTER AEGIS_SRCS EXCLUDE REGEX "target/nxp/frdmrw612/target_cfg\\.c$|load_info_tfm_loader_service\\.c$")
    list(APPEND AEGIS_SRCS ${CMAKE_CURRENT_LIST_DIR}/platform/target_cfg.c)
endif()
set_property(TARGET ${MCUX_SDK_PROJECT_NAME} PROPERTY SOURCES ${AEGIS_SRCS})
